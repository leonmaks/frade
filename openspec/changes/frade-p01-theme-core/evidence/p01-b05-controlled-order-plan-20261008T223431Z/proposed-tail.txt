

// Temporary P01-B05-CONTROLLED-ORDER diagnosis; exact original prefix is restored.
it.each(['no-fit', 'before-narrow', 'narrow-before-resize', 'after-refusal'] as const)(
  'P01-B05-CONTROLLED-ORDER %s exact original restoration', async (schedule) => {
    const fs = await import('node:fs'), path = await import('node:path')
    const output = process.env.FRADE_B05_CONTROLLED_OUTPUT
    if (!output || !path.isAbsolute(output)) throw Error('Missing isolated controlled evidence destination')
    const slices = JSON.parse(fs.readFileSync(process.env.FRADE_B05_VENDOR_SLICES!, 'utf8')) as {
      fit: { functionText: string }
    }
    const f = lowerLayoutFixture(700, 321)
    const fixtureWidth = Object.getOwnPropertyDescriptor(window, 'innerWidth')!.get!
    let viewportOverride: number | undefined
    Object.defineProperty(window, 'innerWidth', { configurable: true,
      get: () => viewportOverride ?? fixtureWidth.call(window) })
    f.root.style.setProperty('left', '65px', '')
    f.root.style.top = '350px'
    const originalStyles = [f.root, f.child].map(node => node.getAttribute('style'))
    for (const node of [f.root, f.child]) {
      for (const [name, axis] of [['offsetLeft', 'left'], ['offsetTop', 'top'],
        ['offsetWidth', 'width'], ['offsetHeight', 'height']] as const) {
        Object.defineProperty(node, name, { configurable: true,
          get: () => node.isConnected ? node.getBoundingClientRect()[axis] : 0 })
      }
    }
    const vendorEnvironment = {
      getDocumentSize: () => ({ width: window.innerWidth, height: window.innerHeight }),
      getDocumentScrollOrigin: () => ({ x: 0, y: 0 }),
    }
    const fit = Function('mxUtils', 'return (' + slices.fit.functionText + ')')(vendorEnvironment) as
      (node: HTMLElement, margin?: number) => void
    const observations: { phase: string; width: number; connected: boolean;
      styles: (string | null)[]; left: string; topPriority: string }[] = []
    const observe = (phase: string) => observations.push({ phase, width: window.innerWidth,
      connected: f.root.isConnected, styles: [f.root, f.child].map(node => node.getAttribute('style')),
      left: f.root.style.left, topPriority: f.root.style.getPropertyPriority('top') })
    let guardStatus = 'NOT_REACHED'
    try {
      observe('before-open')
      await f.open()
      observe('owned-open')
      expect(f.root.style.left).toBe('65px')
      expect(f.root.style.getPropertyPriority('left')).toBe('')
      expect(f.root.style.top).not.toBe('350px')
      expect(f.root.style.getPropertyPriority('top')).toBe('important')
      if (schedule === 'before-narrow') { fit(f.root); observe('vendor-fit-before-narrow') }
      viewportOverride = 88
      observe('viewport-narrow-before-resize-event')
      if (schedule === 'narrow-before-resize') { fit(f.root); observe('vendor-fit-narrow-before-resize') }
      await f.resize(88)
      observe('after-refusal')
      if (schedule === 'after-refusal') { fit(f.root); observe('vendor-fit-after-refusal') }
      await Promise.resolve(); await Promise.resolve()
      expect(f.root.isConnected).toBe(false)
      expect(f.child.isConnected).toBe(false)
      expect(document.activeElement).toBe(f.pages)
      expect(f.pages.getAttribute('aria-expanded')).toBe('false')
      expect(f.replies().filter(reply => reply.status === 'REFUSED')).toHaveLength(1)
      expect(f.effects).toEqual([])
      expect(f.fitting).not.toHaveBeenCalled()
      f.assertSemantic()
      guardStatus = 'PASS'
      observe('before-exact-restoration-assertion')
      expect([f.root, f.child].map(node => node.getAttribute('style'))).toEqual(originalStyles)
    } finally {
      fs.writeFileSync(path.join(output, schedule + '.json'), JSON.stringify({
        schedule, originalStyles, observations, guardStatus, actualFinalStyles:
          [f.root, f.child].map(node => node.getAttribute('style')),
        limitation: 'CONTROLLED_MODEL_ONLY; historical writer/order NOT_PROVEN; no Electron run',
      }, null, 2) + '\n', { flag: 'wx' })
    }
  },
)
