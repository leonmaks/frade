import fs from 'node:fs/promises';
import {resolve,relative,isAbsolute} from 'node:path';
import {hash,packetSnapshot,verifyPacketSource} from './integrity.mjs';
export const CANDIDATE='C:/Users/NVISEN/.codex/worktrees/ui-design-contract/frade';
export const PROCESS_ROOT='C:/Users/NVISEN/.codex/review-automation/packet-v1-20261001T061101Z';
export const RUNTIME='/mnt/c/Users/NVISEN/.codex/review-automation/packet-v1-20261001T061101Z/linux-runtime';
export const linux=p=>'/mnt/'+p[0].toLowerCase()+p.slice(2).replaceAll('\\','/');
export function contained(root,path){const r=relative(resolve(root),resolve(path));if(!r||r.startsWith('..')||isAbsolute(r))throw Error('EXTERNAL_PROCESS_PATH');return resolve(path);}
export function validatePhase(phase){if(!['PRE','POST'].includes(phase))throw Error('INVALID_PHASE');return phase;}
export function assertAllowedArtifact(path,bytes){if(/(^|\/)(\.env($|\.)|\.codex|\.aws|\.ssh|\.git|auth\.json|credentials|id_rsa|id_ed25519|[^/]+\.(pem|key))($|\/)/i.test(path))throw Error('FORBIDDEN_PATH');const text=bytes.toString('utf8');if(/-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----|\bsk-(?:proj-)?[A-Za-z0-9_-]{32,}|\bgh[pousr]_[A-Za-z0-9]{30,}|\bAKIA[A-Z0-9]{16}\b/.test(text))throw Error('SENSITIVE_CONTENT');}
export function configs(){return ['default_permissions="frade-review"','permissions={frade-review={extends=":read-only",filesystem={":root"="deny",":minimal"="read",":workspace_roots"={"."="read"},"'+RUNTIME+'"="read"},network={enabled=false}}}','model_reasoning_effort="xhigh"','approval_policy="never"','web_search="disabled"','apps._default.enabled=false','shell_environment_policy.inherit="none"','shell_environment_policy.set={PATH="/usr/bin:/bin",HOME="/tmp",TMPDIR="/tmp",LANG="C.UTF-8",LC_ALL="C.UTF-8"}'];}
export function parseVerdict(eventsText,report,exitCode){if(exitCode!==0)throw Error('TRANSPORT_NOT_CLEAN');const events=eventsText.split(/\r?\n/).filter(Boolean).map(x=>JSON.parse(x));if(!events.some(e=>e.type==='turn.completed')||events.some(e=>e.type==='turn.failed'||e.type==='error'))throw Error('INCOMPLETE_EVENT_STREAM');const verdicts=[...report.replace(/\\+_/g,'_').matchAll(/^\s*GATE_STATUS:\s*(PASS|FAIL)\s*$/gm)];if(verdicts.length!==1)throw Error('AMBIGUOUS_GATE');return {gateStatus:verdicts[0][1],threadId:events.find(e=>e.type==='thread.started')?.thread_id??null};}
export async function verifyPrepared(candidate,prepared,phase){validatePhase(phase);if(await fs.realpath(candidate)!==await fs.realpath(CANDIDATE))throw Error('UNAUTHORIZED_CANDIDATE');contained(PROCESS_ROOT,prepared);const packet=resolve(prepared,'packet'),manifest=JSON.parse(await fs.readFile(resolve(packet,'REVIEW-PACKET-MANIFEST.json'),'utf8'));if(manifest.phase!==phase)throw Error('PHASE_MISMATCH');if(!manifest.scope||!manifest.authorization)throw Error('MISSING_REVIEW_AUTHORITY');await verifyPacketSource(candidate,packet,manifest);const actual=await packetSnapshot(packet),expected=new Set([...manifest.artifacts.map(a=>a.path),'REVIEW-PACKET-MANIFEST.json']);if(actual.artifacts.length!==expected.size||actual.artifacts.some(a=>!expected.has(a.path)))throw Error('UNDECLARED_PACKET_FILE');for(const a of manifest.artifacts){const bytes=await fs.readFile(resolve(packet,a.path));assertAllowedArtifact(a.path,bytes);}return {packet,manifest,packetDigest:actual.digest,manifestSha256:hash(await fs.readFile(resolve(packet,'REVIEW-PACKET-MANIFEST.json')))};}
